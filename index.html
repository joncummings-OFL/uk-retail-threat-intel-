# ==============================================================================
# UK RETAIL CYBER THREAT INTELLIGENCE (CTI) SCRAPER & DAILY SUMMARY PIPELINE
# Dependencies: pip install feedparser requests beautifulsoup4 google-genai
# ==============================================================================

import os
import re
import json
import logging
from datetime import datetime, timezone
import feedparser
import requests
from bs4 import BeautifulSoup
from google import genai

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")

# 1. CURATED SOURCES (UK + Global Cybersecurity feeds)
FEEDS = [
    {"name": "NCSC UK Advisories", "url": "https://www.ncsc.gov.uk/api/1/services/v1/report-rss-feed.xml", "category": "UK Govt"},
    {"name": "The Register Security (UK)", "url": "https://www.theregister.com/security/headlines.atom", "category": "Tech News"},
    {"name": "BleepingComputer", "url": "https://www.bleepingcomputer.com/feed/", "category": "Global Alerts"},
    {"name": "Dark Reading", "url": "https://www.darkreading.com/rss.xml", "category": "Vulnerabilities"},
    {"name": "SecurityWeek", "url": "https://feeds.feedburner.com/securityweek", "category": "Enterprise"},
]

# 2. UK RETAIL TAXONOMY & KEYWORDS
RETAIL_KEYWORDS = [
    r"\bretail\b", r"\be-commerce\b", r"\becommerce\b", r"\bpoint of sale\b", r"\bpos\b",
    r"\bmagecart\b", r"\be-skimming\b", r"\bcheckout\b", r"\bpci[- ]dss\b", r"\bloyalty\b",
    r"\bcredential stuffing\b", r"\baccount takeover\b", r"\bato\b", r"\bgift card\b",
    r"\bsupply chain\b", r"\bwarehouse\b", r"\blogistics\b", r"\bico\b", r"\bshop\b",
    r"\btesco\b", r"\bsainsbury\b", r"\basda\b", r"\bmarks & spencer\b", r"\bm&s\b",
    r"\bboots\b", r"\bcurrys\b", r"\bnext\b", r"\bjohn lewis\b", r"\bharrods\b"
]
KEYWORD_PATTERN = re.compile("|".join(RETAIL_KEYWORDS), re.IGNORECASE)

def fetch_and_filter_intel():
    """Scrapes RSS feeds and extracts items relevant to UK retail cyber posture."""
    matched_articles = []
    
    for feed in FEEDS:
        logging.info(f"Ingesting: {feed['name']}...")
        try:
            parsed = feedparser.parse(feed["url"])
            for entry in parsed.entries[:15]:  # inspect latest 15
                title = entry.get("title", "")
                summary = entry.get("summary", "") or entry.get("description", "")
                link = entry.get("link", "")
                
                # Strip HTML from summary for clean matching
                clean_summary = BeautifulSoup(summary, "html.parser").get_text()
                combined_text = f"{title} {clean_summary}"
                
                # Check for retail relevance
                matches = KEYWORD_PATTERN.findall(combined_text)
                if matches:
                    matched_articles.append({
                        "source": feed["name"],
                        "title": title,
                        "summary": clean_summary[:350] + "...",
                        "link": link,
                        "matched_terms": list(set([m.lower() for m in matches])),
                        "published": entry.get("published", str(datetime.now(timezone.utc)))
                    })
        except Exception as e:
            logging.error(f"Error scraping {feed['name']}: {e}")

    logging.info(f"Retrieved {len(matched_articles)} retail-relevant threat reports.")
    return matched_articles

def generate_ciso_daily_brief(articles):
    """Uses Gemini to synthesize an executive threat summary for UK retail."""
    if not articles:
        return "No high-priority retail threats detected in the last cycle."

    # Gemini API client initialization
    # Ensure GEMINI_API_KEY is set in environment: export GEMINI_API_KEY="your-api-key"
    client = genai.Client()

    prompt = f"""
    Act as a Principal Cyber Threat Intelligence Analyst for the UK Retail Sector.
    Review the following raw scraped cyber security news and indicators collected today:

    {json.dumps(articles, indent=2)}

    Generate a high-impact Daily Cyber Threat Intelligence Executive Briefing containing:
    1. **Executive TL;DR**: 2-3 sentences on the prevailing UK retail threat landscape.
    2. **Critical Incidents & Exploits**: Highlight top vulnerabilities (e.g. Magecart, POS, Supply Chain, ATO).
    3. **UK Regulatory & Compliance Impact**: Potential UK GDPR / ICO / PCI-DSS v4.0 implications.
    4. **Tactical MITRE ATT&CK Mapping**: Key tactics observed (Initial Access, Persistence, Exfiltration).
    5. **Immediate Defensive Recommendations for Retail CISOs**: Actionable steps for SOC and e-comm engineering.

    Format cleanly in professional Markdown with bullet points.
    """

    response = client.models.generate_content(
        model="gemini-3-flash-preview",
        contents=prompt
    )
    return response.text

def notify_channels(summary_text):
    """Dispatches briefing to Slack or email."""
    slack_webhook = os.getenv("SLACK_WEBHOOK_URL")
    if slack_webhook:
        requests.post(slack_webhook, json={"text": f"🛡️ *UK Retail Daily CTI Briefing*\n\n{summary_text}"})
        logging.info("Sent summary to Slack.")
    else:
        logging.info("Slack webhook not configured. Printing to stdout.\n")
        print(summary_text)

if __name__ == "__main__":
    logging.info("Starting Daily UK Retail Threat Intelligence Collector...")
    intel_items = fetch_and_filter_intel()
    briefing = generate_ciso_daily_brief(intel_items)
    notify_channels(briefing)
